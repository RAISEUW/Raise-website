    ```
    export async function downloadToFile(url, destPath) {
      const res = await fetch(url, { headers: { 'User-Agent': UA } });
      if (!res.ok) throw new Error(`${url}: HTTP ${res.status}`);
      await mkdir(path.dirname(destPath), { recursive: true });
      await pipeline(Readable.fromWeb(res.body), createWriteStream(destPath));
    }

    export function extFromUrl(urlStr, fallback) {
      try {
        const urlPath = new URL(urlStr).pathname;
        const m = urlPath.match(/\.(jpg|jpeg|png|webp|gif|pdf)$/i);
        return m ? m[0].toLowerCase() : fallback;
      } catch {
        return fallback;
      }
    }
    ```

    ```
    import matter from 'gray-matter';
    import { readFile, writeFile, readdir } from 'node:fs/promises';
    import path from 'node:path';
    import pLimit from 'p-limit';
    import { downloadToFile, extFromUrl } from '../lib/assets.mjs';
    import { cleanFrontmatter } from '../lib/mdx.mjs';

    const PEOPLE_DIR = 'src/content/people';
    const PUBS_DIR = 'src/content/publications';
    const IMG_DIR = 'public/images/people';
    const PDF_DIR = 'public/pdfs';

    // Detect external URL (http/https) vs local path (starts with /)
    function isExternalUrl(s) { return typeof s === 'string' && /^https?:\/\//i.test(s); }

    export async function downloadAssets() {
      const limit = pLimit(5);
      const jobs = [];
      let imgOk = 0, imgFail = 0, pdfOk = 0, pdfFail = 0, pdfSkipped = 0;

      // --- PEOPLE: headshots ---
      const peopleFiles = (await readdir(PEOPLE_DIR)).filter(f => f.endsWith('.mdx'));
      for (const file of peopleFiles) {
        const p = path.join(PEOPLE_DIR, file);
        const raw = await readFile(p, 'utf8');
        const parsed = matter(raw);
        const photoVal = parsed.data.photo;

        // If photo already looks like a local path, skip download but keep path.
        // If photo is empty, nothing to do.
        if (!photoVal) continue;

        // Plan 02-02 emitted photo as `/images/people/{slug}.{ext}` — but the actual URL
        // lives in the scraped DOM. The parsers discarded it. Workaround: if we still have
        // the raw URL (either from re-scraping or from a sidecar), download it.
        //
        // Strategy: re-fetch /people/, re-extract the image URL for this person by name,
        // and download it. This avoids adding sidecar state.

        // See below: we build a name→URL map first, then iterate.
      }

      // Re-scrape /people/ to build name→photoURL map (cheap: one HTTP GET).
      const cheerio = await import('cheerio');
      const peopleRes = await fetch('https://raise.uw.edu/people/', { headers: { 'User-Agent': 'RAISE-content-migration/1.0' } });
      if (!peopleRes.ok) throw new Error(`people page: HTTP ${peopleRes.status}`);
      const $ = cheerio.load(await peopleRes.text());

      const nameToPhoto = new Map();
      $('.et_pb_team_member').each((_, tm) => {
        const $tm = $(tm);
        const name = $tm.find('.et_pb_module_header').text().trim();
        const url = $tm.closest('.et_pb_row').find('.et_pb_image img').first().attr('src')
          || $tm.find('img').first().attr('src');
        if (name && url) nameToPhoto.set(name, url);
      });

      // Now: for each person MDX, download headshot if photo field is set.
      for (const file of peopleFiles) {
        const p = path.join(PEOPLE_DIR, file);
        const raw = await readFile(p, 'utf8');
        const parsed = matter(raw);
        const name = parsed.data.name;
        const url = nameToPhoto.get(name);
        if (!url) continue;

        const ext = extFromUrl(url, '.jpg');
        const slug = file.replace(/\.mdx$/, '');
        const dest = path.join(IMG_DIR, slug + ext);
        const localPath = `/images/people/${slug}${ext}`;

        jobs.push(limit(async () => {
          try {
            await downloadToFile(url, dest);
            // Rewrite frontmatter photo field to local path (already was, but ensure ext matches).
            parsed.data.photo = localPath;
            await writeFile(p, matter.stringify(parsed.content, cleanFrontmatter(parsed.data)), 'utf8');
            imgOk++;
          } catch (err) {
            console.warn(`  photo failed for ${name}: ${err.message}`);
            // Omit photo field on failure (D-03)
            delete parsed.data.photo;
            await writeFile(p, matter.stringify(parsed.content, cleanFrontmatter(parsed.data)), 'utf8');
            imgFail++;
          }
        }));
      }

      // --- PUBLICATIONS: PDFs ---
      // Scope (per Pitfall #4): download PDFs only when (a) pdf field already ends in .pdf,
      // or (b) pdf field is an arxiv.org/pdf/... URL (converted in 02-01 parser).
      // Publications that link to DOI records or publisher landing pages are INTENTIONALLY
      // left as external links — the detail page renders them as "Read on <publisher>" or
      // via the doi: field. This is the accepted reality of MIG-05 (see <objective> scope note).
      const pubFiles = (await readdir(PUBS_DIR)).filter(f => f.endsWith('.mdx'));
      for (const file of pubFiles) {
        const p = path.join(PUBS_DIR, file);
        const raw = await readFile(p, 'utf8');
        const parsed = matter(raw);
        const pdfVal = parsed.data.pdf;

        if (!pdfVal) { pdfSkipped++; continue; }
        if (!isExternalUrl(pdfVal)) { pdfSkipped++; continue; }  // already local
        if (!/\.pdf(\?|$)/i.test(pdfVal) && !/arxiv\.org\/pdf/i.test(pdfVal)) {
          // Not a direct PDF — leave as external link in MDX (detail page can still render)
          pdfSkipped++;
          continue;
        }

        const slug = file.replace(/\.mdx$/, '');
        const dest = path.join(PDF_DIR, slug + '.pdf');
        const localPath = `/pdfs/${slug}.pdf`;

        jobs.push(limit(async () => {
          try {
            await downloadToFile(pdfVal, dest);
            parsed.data.pdf = localPath;
            await writeFile(p, matter.stringify(parsed.content, cleanFrontmatter(parsed.data)), 'utf8');
            pdfOk++;
          } catch (err) {
            console.warn(`  pdf failed for ${file}: ${err.message}`);
            // Leave pdf field as external URL — still usable (detail page follows link)
            pdfFail++;
          }
        }));
      }

      await Promise.all(jobs);
      return { imgOk, imgFail, pdfOk, pdfFail, pdfSkipped };
    }
    ```

    **A. Create site/scripts/parsers/links.mjs** — exact case-insensitive name match between person.name and publication.authors + event.speaker:

    ```
    import matter from 'gray-matter';
    import { readFile, writeFile, readdir } from 'node:fs/promises';
    import path from 'node:path';
    import { cleanFrontmatter } from '../lib/mdx.mjs';

    const PEOPLE_DIR = 'src/content/people';
    const PUBS_DIR = 'src/content/publications';
    const EVENTS_DIR = 'src/content/events';

    async function readCollection(dir) {
      const files = (await readdir(dir)).filter(f => f.endsWith('.mdx'));
      const entries = [];
      for (const f of files) {
        const raw = await readFile(path.join(dir, f), 'utf8');
        const parsed = matter(raw);
        entries.push({ id: f.replace(/\.mdx$/, ''), data: parsed.data, content: parsed.content, path: path.join(dir, f) });
      }
      return entries;
    }

    function nameKey(s) { return (s ?? '').toLowerCase().trim().replace(/\s+/g, ' '); }

    export async function wireLinks() {
      const people = await readCollection(PEOPLE_DIR);
      const pubs = await readCollection(PUBS_DIR);
      const events = await readCollection(EVENTS_DIR);

      let pubLinks = 0, talkLinks = 0;

      for (const person of people) {
        const personKey = nameKey(person.data.name);
        const linkedPubs = pubs
          .filter(p => (p.data.authors ?? []).some(a => nameKey(a) === personKey))
          .map(p => p.id);
        const linkedTalks = events
          .filter(e => nameKey(e.data.speaker) === personKey)
          .map(e => e.id);

        if (linkedPubs.length === 0 && linkedTalks.length === 0
            && (person.data.linkedPublications ?? []).length === 0
            && (person.data.linkedTalks ?? []).length === 0) {
          continue;  // nothing to write
        }

        person.data.linkedPublications = linkedPubs;
        person.data.linkedTalks = linkedTalks;

        await writeFile(person.path, matter.stringify(person.content, cleanFrontmatter(person.data)), 'utf8');
        pubLinks += linkedPubs.length;
        talkLinks += linkedTalks.length;
      }

      return { pubLinks, talkLinks };
    }
    ```
