    **A. site/scripts/lib/mdx.mjs** — MDX writer with D-03 spread-conditional pattern:

    ```
    import matter from 'gray-matter';
    import { writeFile, mkdir } from 'node:fs/promises';
    import path from 'node:path';

    // Strip undefined/null/empty-string values so D-03 (omit-when-missing) is enforced.
    // Use spread-conditional at call sites: ...(val && { field: val })
    export function cleanFrontmatter(obj) {
      const out = {};
      for (const [k, v] of Object.entries(obj)) {
        if (v === undefined || v === null) continue;
        if (typeof v === 'string' && v.trim() === '') continue;
        if (Array.isArray(v) && v.length === 0 && k !== 'researchAreas' && k !== 'topics') continue;
        out[k] = v;
      }
      return out;
    }

    export async function writeEntry({ dir, filename, frontmatter, body = '' }) {
      const clean = cleanFrontmatter(frontmatter);
      const mdx = matter.stringify(body, clean);
      await mkdir(dir, { recursive: true });
      await writeFile(path.join(dir, filename), mdx, 'utf8');
    }
    ```

    **B. site/scripts/lib/slugify.mjs** — kebab-case with collision handling:

    ```
    import slug from 'slug';

    const seen = new Map();  // slug to count

    // Keep slugs short: take first 6 words of title, slugify, dedupe with -2/-3 suffixes.
    export function makeSlug(title, hint) {
      const base = slug(title.split(/\s+/).slice(0, 6).join(' '), { lower: true });
      const candidate = hint ? `${base}-${slug(hint, { lower: true })}` : base;
      const count = (seen.get(candidate) ?? 0) + 1;
      seen.set(candidate, count);
      return count === 1 ? candidate : `${candidate}-${count}`;
    }

    export function resetSlugCache() { seen.clear(); }
    ```

    **C. site/scripts/lib/dates.mjs** — multi-format parse with fallbacks:

    ```
    import { parse as parseFns, isValid } from 'date-fns';

    const FORMATS = [
      'd MMMM yyyy',      // "7 June 2025"
      'd MMM yyyy',       // "8 Apr 2025"
      'MMMM d, yyyy',     // "April 19, 2023"
      'MMMM dd, yyyy',    // "October 08, 2023"
      'MMM d yyyy',       // event fallback after ordinal strip
      'MMM dd yyyy',
      'yyyy-MM-dd',
    ];

    export function parseDate(raw) {
      if (!raw) return undefined;
      // Strip ordinal suffixes (1st, 2nd, 3rd, 4th, AND the "3th" typo from source)
      const cleaned = raw.replace(/(\d+)(st|nd|rd|th)/gi, '$1').trim();
      for (const fmt of FORMATS) {
        const d = parseFns(cleaned, fmt, new Date());
        if (isValid(d)) return d;
      }
      return undefined;  // omit when unparseable (D-03)
    }
    ```

    **D. site/scripts/lib/topics.mjs** — pipe-separated normalizer with acronym whitelist:

    ```
    const ACRONYMS = new Set(['LLM', 'AI', 'NLP', 'HCI', 'HPC', 'COVID-19', 'DOI', 'API', 'URL', 'PDF', 'LM', 'ML', 'RAG', 'GPT']);

    function titleCaseToken(word) {
      const upper = word.toUpperCase();
      if (ACRONYMS.has(upper)) return upper;
      return word.charAt(0).toUpperCase() + word.slice(1).toLowerCase();
    }

    export function normalizeTopic(raw) {
      return raw.trim().split(/\s+/).map(titleCaseToken).join(' ');
    }

    export function splitTopics(pipeString) {
      return pipeString.split('|').map(s => normalizeTopic(s.trim())).filter(Boolean);
    }
    ```

    ```
    import path from 'node:path';
    import { rm } from 'node:fs/promises';
    import { writeEntry } from '../lib/mdx.mjs';
    import { makeSlug, resetSlugCache } from '../lib/slugify.mjs';
    import { parseDate } from '../lib/dates.mjs';
    import { splitTopics } from '../lib/topics.mjs';

    const OUT_DIR = 'src/content/publications';

    // Heuristic: split "VENUE, DATE" on the LAST comma where right side looks like a date.
    // Examples: "Proceedings of ... Media, 7 June 2025" → venue="Proceedings...", date="7 June 2025"
    function splitVenueDate(strongText) {
      const text = strongText.trim();
      // Regex captures trailing date: (d MMMM yyyy | d MMM yyyy | MMMM d, yyyy | MMMM dd, yyyy)
      const m = text.match(/^(.*?),?\s+(\d{1,2}\s+\w+\s+\d{4}|\w+\s+\d{1,2},\s*\d{4})\s*$/);
      if (m) return [m[1].replace(/,\s*$/, '').trim(), m[2].trim()];
      return [text, undefined];
    }

    // Extract DOI from doi.org URLs (populate `doi:` field when `pdf:` isn't local).
    function extractDoi(url) {
      if (!url) return undefined;
      const m = url.match(/doi\.org\/(10\.[^\s?#]+)/i);
      return m ? m[1] : undefined;
    }

    export async function parsePublications($) {
      resetSlugCache();
      await rm(OUT_DIR, { recursive: true, force: true });

      let count = 0;
      // Select on stable token (.et_pb_blurb) — NEVER numbered variants like .et_pb_blurb_0_tb_body
      const blurbs = $('.et_pb_blurb').toArray();

      for (const el of blurbs) {
        const $el = $(el);
        const title = $el.find('.et_pb_module_header span').first().text().trim()
          || $el.find('.et_pb_module_header').first().text().trim();
        if (!title) continue;

        // venue + date live in the first <strong> of the description
        const strongText = $el.find('.et_pb_blurb_description strong').first().text().trim();
        const [venue, dateStr] = splitVenueDate(strongText);
        const date = parseDate(dateStr);

        // Authors are in a <p> following the strong; take the first <p> that has no <strong>
        const ps = $el.find('.et_pb_blurb_description p').toArray();
        let authorsText = '';
        for (const p of ps) {
          const $p = $(p);
          if ($p.find('strong').length === 0 && $p.text().trim()) {
            authorsText = $p.text().trim();
            break;
          }
        }
        const authors = authorsText.split(/,\s*/).map(s => s.trim()).filter(Boolean);

        // Sibling et_pb_buttons inside the parent row: "Read Here" link + pipe-separated topic button
        const $row = $el.closest('.et_pb_row');
        const $buttons = $row.find('.et_pb_button');
        let readHere, topicText;
        $buttons.each((_, b) => {
          const $b = $(b);
          const text = $b.text().trim();
          const href = $b.attr('href');
          if (text === 'Read Here') readHere = href;
          else if (text && text !== 'Read Here') topicText = text;
        });
        const topics = topicText ? splitTopics(topicText) : [];

        // Decision (Pitfall #4): if arXiv abs URL → convert to PDF URL for later download (Plan 02-03);
        // for now, store `pdf` if it ends in .pdf or the arXiv conversion applies; else store `doi`.
        let pdf, doi;
        if (readHere) {
          if (/\.pdf(\?|$)/i.test(readHere)) {
            pdf = readHere;  // external PDF URL; Plan 02-03 downloads and rewrites to /pdfs/{slug}.pdf
          } else if (/arxiv\.org\/abs\//i.test(readHere)) {
            pdf = readHere.replace('/abs/', '/pdf/').replace(/(\.pdf)?$/, '.pdf');
          } else {
            doi = extractDoi(readHere) ?? readHere;  // DOI ID if doi.org, else raw landing-page URL
          }
        }

        // D-03: spread-conditional omits undefined fields cleanly (cleanFrontmatter also strips).
        const frontmatter = {
          title,
          authors,
          venue,
          ...(date && { date }),
          topics,
          ...(pdf && { pdf }),
          ...(doi && { doi }),
        };

        const filename = makeSlug(title) + '.mdx';
        await writeEntry({ dir: OUT_DIR, filename, frontmatter, body: '' });
        count++;
      }
      return count;
    }
    ```
